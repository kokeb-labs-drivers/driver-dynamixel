475constexpr Error Validate(const Config& config) noexcept {
476 const bool enums_ok =
477 detail::IsEnumerator(config.protocol) && detail::IsEnumerator(config.echo) &&
478 detail::IsEnumerator(config.alert_policy) && detail::IsEnumerator(config.status_return);
479 const bool baud_ok = (config.baud >= kMinBaud) && (config.baud <= kMaxBaud);
480 const bool timeouts_ok =
481 detail::InRange(config.reply_timeout_us, kMinTimeoutUs, kMaxTimeoutUs) &&
482 detail::InRange(config.gap_timeout_us, kMinTimeoutUs, kMaxTimeoutUs) &&
483 detail::InRange(config.broadcast_timeout_us, kMinTimeoutUs, kMaxTimeoutUs);
484 const bool waits_ok =
485 (config.reboot_wait_us <= kMaxSettleUs) && (config.baud_change_wait_us <= kMaxSettleUs);
486 const bool retries_ok = config.retries <= kMaxRetries;
487 return (enums_ok && baud_ok && timeouts_ok && waits_ok && retries_ok) ? Error::Ok
488 : Error::InvalidArgument;
489}

504constexpr ReplyInfo Classify(Protocol protocol, uint8_t id, uint8_t error_byte) noexcept {
505 ReplyInfo info{};
506 info.id = id;
507 info.error_byte = error_byte;
508 const auto bits = static_cast<uint32_t>(error_byte);
509 if (protocol == Protocol::V2) {
510 info.alert = (bits & kAlertBit) != 0U;
511 info.rejected = (bits & 0x7FU) != 0U;
512 } else {
513 info.alert = (bits & p1_error::kConditions) != 0U;
514 info.rejected = (bits & p1_error::kCommandFaults) != 0U;
515 }
516 return info;
517}

525constexpr Protocol2Error ErrorNumber(uint8_t error_byte) noexcept {
526 const uint32_t number = static_cast<uint32_t>(error_byte) & 0x7FU;
527 return (number <= 7U) ? static_cast<Protocol2Error>(number) : Protocol2Error::ResultFail;
528}
