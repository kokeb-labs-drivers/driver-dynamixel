958[[nodiscard]] constexpr bool BaudValueFor(Family family, uint32_t bits_per_second,
959 uint8_t& value) noexcept {
960 std::size_t count = 0;
961 switch (Info(family).baud_scheme) {
962 case BaudScheme::Divider:
963 for (uint32_t n = 0U; n <= 254U; ++n) {
964 if (bits_per_second == (2'000'000U / (n + 1U))) {
965 value = static_cast<uint8_t>(n);
966 return true;
967 }
968 }
969 return false;
970 case BaudScheme::Table4:
971 count = 4;
972 break;
973 case BaudScheme::Table7:
974 count = 7;
975 break;
976 case BaudScheme::Table8:
977 count = 8;
978 break;
979 case BaudScheme::None:
980 break;
981 default:
982 break;
983 }
984 for (std::size_t i = 0; i < count; ++i) {
985 if (kRates[i] == bits_per_second) {
986 value = static_cast<uint8_t>(i);
987 return true;
988 }
989 }
990 return false;
991}
