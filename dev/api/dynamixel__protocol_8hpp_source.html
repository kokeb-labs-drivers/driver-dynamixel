71constexpr std::array<uint16_t, 256> MakeTable() noexcept {
72 std::array<uint16_t, 256> table{};
73 for (uint32_t i = 0U; i < 256U; ++i) {
74 uint32_t crc = i << 8U;
75 for (uint32_t bit = 0U; bit < 8U; ++bit) {
76 crc = ((crc & 0x8000U) != 0U) ? (((crc << 1U) ^ 0x8005U) & 0xFFFFU) : ((crc << 1U) & 0xFFFFU);
77 }
78 table[i] = static_cast<uint16_t>(crc);

92[[nodiscard]] constexpr uint16_t Crc16(uint16_t crc, std::span<const uint8_t> data) noexcept {
93 uint32_t acc = crc;
94 for (const uint8_t byte : data) {
95 const uint32_t index = ((acc >> 8U) ^ static_cast<uint32_t>(byte)) & 0xFFU;
96 acc = ((acc << 8U) & 0xFFFFU) ^ static_cast<uint32_t>(detail_crc::kTable[index]);
97 }
98 return static_cast<uint16_t>(acc);
99}

106[[nodiscard]] constexpr uint8_t Checksum1(std::span<const uint8_t> data) noexcept {
107 uint32_t sum = 0U;

192 [[nodiscard]] std::size_t Finish() noexcept {
193 if (protocol_ == Protocol::V2) {
194 const std::size_t length = (len_ - params_start_) + 3U;
195 if ((length > 0xFFFFU) || ((len_ + 2U) > buf_.size()) || overflow_) {
196 return 0U;
197 }
198 buf_[5] = static_cast<uint8_t>(length & 0xFFU);
199 buf_[6] = static_cast<uint8_t>((length >> 8U) & 0xFFU);
200 const uint16_t crc = Crc16(0_u16, std::span<const uint8_t>{buf_.data(), len_});
201 buf_[len_] = static_cast<uint8_t>(static_cast<uint32_t>(crc) & 0xFFU);
202 buf_[len_ + 1U] = static_cast<uint8_t>((static_cast<uint32_t>(crc) >> 8U) & 0xFFU);
203 return len_ + 2U;
204 }
205 const std::size_t params = len_ - params_start_;
206 if ((params > kMaxParamsV1) || ((len_ + 1U) > buf_.size()) || overflow_) {
207 return 0U;
208 }
209 buf_[3] = static_cast<uint8_t>(params + 2U);
210 buf_[len_] = Checksum1(std::span<const uint8_t>{buf_.data() + 2, len_ - 2U});
211 return len_ + 1U;
212 }

216 if (len_ < buf_.size()) {
217 buf_[len_] = byte;
218 ++len_;
219 } else {
220 overflow_ = true;
221 }

240[[nodiscard]] constexpr std::size_t Destuff(std::span<uint8_t> params) noexcept {
241 std::size_t out = 0;
242 uint32_t run = 0U;
243 for (std::size_t i = 0; i < params.size(); ++i) {
244 const uint8_t byte = params[i];
245 params[out] = byte;
246 ++out;
247 if (byte == 0xFF_u8) {
248 run = (run < 2U) ? (run + 1U) : 2U;
249 } else if ((byte == 0xFD_u8) && (run == 2U)) {
250 // FF FF FD must be followed by the stuffed FD, which is dropped.
251 if (((i + 1U) >= params.size()) || (params[i + 1U] != 0xFD_u8)) {
252 return kBadStuffing;
253 }
254 ++i;
255 run = 0U;
256 } else {
257 run = 0U;
258 }
259 }
260 return out;
261}

319constexpr bool IsCandidate(Protocol protocol, std::span<const uint8_t> bytes, std::size_t pos,
320 std::size_t left) noexcept {
321 if (protocol == Protocol::V2) {

340constexpr bool IsPlausible(Protocol protocol, std::span<const uint8_t> bytes, std::size_t pos,
341 std::size_t left, std::size_t max_frame, std::size_t& total) noexcept {
342 if (protocol == Protocol::V2) {
343 const std::size_t length = static_cast<std::size_t>(bytes[pos + 5U]) |
344 (static_cast<std::size_t>(bytes[pos + 6U]) << 8U);
345 total = 7U + length;
346 const bool wrong_instruction =
347 (left >= 8U) && (bytes[pos + 7U] != static_cast<uint8_t>(Instruction::Status));
348 return (length >= 4U) && (total <= max_frame) && !wrong_instruction;
349 }
350 const std::size_t length = bytes[pos + 3U];
351 total = 4U + length;
352 return (length >= 2U) && (total <= max_frame) && (bytes[pos + 2U] <= kMaxIdV1);
353}

361constexpr bool IsChecked(Protocol protocol, std::span<const uint8_t> frame) noexcept {
362 const std::size_t total = frame.size();
363 if (protocol == Protocol::V2) {
364 const uint16_t crc = Crc16(0_u16, frame.first(total - 2U));
365 return (frame[total - 2U] == static_cast<uint8_t>(static_cast<uint32_t>(crc) & 0xFFU)) &&
366 (frame[total - 1U] == static_cast<uint8_t>((static_cast<uint32_t>(crc) >> 8U) & 0xFFU));
367 }
368 return frame[total - 1U] == Checksum1(frame.subspan(2U, total - 3U));
369}

386[[nodiscard]] constexpr ScanResult Scan(Protocol protocol, std::span<const uint8_t> bytes,
387 std::size_t max_frame) noexcept {
388 ScanResult result{};
389 const std::size_t header = (protocol == Protocol::V2) ? 4U : 2U;
390 const std::size_t min_for_len = (protocol == Protocol::V2) ? 7U : 4U;
391 std::size_t pos = 0;
392 while (pos < bytes.size()) {
393 const std::size_t left = bytes.size() - pos;
394 // A partial header at the end of the data may still become one: keep it.
395 if ((left < header) && detail_scan::IsPartialHeader(protocol, bytes, pos, left)) {
396 result.discard = pos;
397 result.need = header - left;
398 return result;
399 }
400 if ((left < header) || !detail_scan::IsCandidate(protocol, bytes, pos, left)) {
401 result.garbage += 1U;
402 ++pos;
403 continue;
404 }
405 if (left < min_for_len) {
406 result.discard = pos;
407 result.need = min_for_len - left;
408 return result;
409 }
410 std::size_t total = 0;
411 if (!detail_scan::IsPlausible(protocol, bytes, pos, left, max_frame, total)) {
412 result.garbage += 1U;
413 ++pos;
414 continue;
415 }
416 if (left < total) {
417 result.discard = pos;
418 result.need = total - left;
419 return result;
420 }
421 if (detail_scan::IsChecked(protocol, bytes.subspan(pos, total))) {
422 result.kind = ScanResult::Kind::Frame;
423 result.discard = pos;
424 result.frame_size = total;
425 return result;
426 }
427 result.crc_drops += 1U;
428 ++pos;
429 }
430 result.discard = pos;
431 return result;
432}

483[[nodiscard]] constexpr int32_t DecodeValue(Kind kind, std::span<const uint8_t> data) noexcept {
484 const uint32_t raw = LoadLe(data);
485 const uint32_t bits = static_cast<uint32_t>(data.size()) * 8U;
486 switch (kind) {
487 case Kind::Signed:
488 if ((bits < 32U) && ((raw & (1U << (bits - 1U))) != 0U)) {
489 return static_cast<int32_t>(raw) - static_cast<int32_t>(1U << bits);
490 }
491 return static_cast<int32_t>(raw);
492 case Kind::SignMag10: {
493 const auto magnitude = static_cast<int32_t>(raw & 0x3FFU);
494 return ((raw & 0x400U) != 0U) ? -magnitude : magnitude;
495 }
496 case Kind::Offset2048:
497 return static_cast<int32_t>(raw) - 2048;
498 case Kind::Unsigned:
499 return static_cast<int32_t>(raw);
500 default:
501 break;
502 }
503 return static_cast<int32_t>(raw);
504}

514[[nodiscard]] constexpr bool EncodeValue(Kind kind, std::size_t size, int32_t value,
515 std::span<uint8_t, 4> out) noexcept {
516 if ((size != 1U) && (size != 2U) && (size != 4U)) {
517 return false;
518 }
519 const int64_t v = value;
520 const uint32_t bits = static_cast<uint32_t>(size) * 8U;
521 const auto range = static_cast<int64_t>(uint64_t{1} << bits); // 2^bits
522 const int64_t half = range / 2;
523 uint32_t raw = 0U;
524 switch (kind) {
525 case Kind::Unsigned:
526 if ((v < 0) || (v >= range)) {
527 return false;
528 }
529 raw = static_cast<uint32_t>(v);
530 break;
531 case Kind::Signed:
532 if ((v < -half) || (v >= half)) {
533 return false;
534 }
535 raw = static_cast<uint32_t>(static_cast<uint64_t>(v) & 0xFFFFFFFFULL);
536 break;
537 case Kind::SignMag10:
538 if ((v < -1023) || (v > 1023)) {
539 return false;
540 }
541 raw = (v < 0) ? (static_cast<uint32_t>(-v) | 0x400U) : static_cast<uint32_t>(v);
542 break;
543 case Kind::Offset2048:
544 if (((v + 2048) < 0) || ((v + 2048) >= range)) {
545 return false;
546 }
547 raw = static_cast<uint32_t>(v + 2048);
548 break;
549 default:
550 return false;
551 }
552 for (std::size_t i = 0; i < size; ++i) {
553 out[i] = static_cast<uint8_t>((raw >> (8U * static_cast<uint32_t>(i))) & 0xFFU);
554 }
555 return true;
556}
