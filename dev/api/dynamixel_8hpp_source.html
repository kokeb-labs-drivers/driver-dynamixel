39// Optional hooks: a `Declares*` concept only tests the name, the `Has*` concept the exact
40// signature, `noexcept` and the result type; the driver rejects a hook that is declared but does
41// not match (KB-CPP20-007).
42
44template <typename Port>
45concept DeclaresSetBaudRate = requires(Port& port) { port.SetBaudRate(1_u32); };
46
48template <typename Port>
49concept HasSetBaudRate =
50 requires { ExactHook<Port, bool(uint32_t) noexcept>::Take(&Port::SetBaudRate); };
51
53template <typename Port>
54concept DeclaresLog = requires(Port& port) { port.Log(LogLevel::Info, ""); };
55
57template <typename Port>
58concept HasLog =
59 requires { ExactHook<Port, void(LogLevel, const char*) noexcept>::Take(&Port::Log); };
60
62template <typename Port>
63void Log(Port& port, [[maybe_unused]] LogLevel level,
64 [[maybe_unused]] const char* message) noexcept {
65 if constexpr (HasLog<Port>) {
66 port.Log(level, message);
67 } else {
68 static_cast<void>(port);
69 }
70}
